# Extension sections are separated by ; and support several entries
#   default=name:value,name:value  default-state overrides, listing only properties that differ from vanilla any()
#     maps to registerDefaultState at the end of the vanilla constructor; without it the default state falls back to any()

#   place=category          general placement-facing implementation category
#     walks the class chain to the first class that defines getStateForPlacement; only the vanilla generic implementations can be copied
#   occlude=false        the block does not block light, maps to vanilla Properties.noOcclusion
#   conductor=false/true redstone conductor special case, maps to vanilla Properties.isRedstoneConductor
#     vanilla defaults to checking whether the collision shape fills a full block; glass and leaves fill it yet set never, soul sand and mud do not fill it yet set always

#   blast=value          explosion resistance, maps to vanilla Properties.explosionResistance
#     vanilla strength(x) means strength(x, x), so a single-argument strength sets the resistance as well
#     only values above zero are written; anything unlisted keeps the vanilla default of 0
